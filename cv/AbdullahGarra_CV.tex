\section{Research Experience}
\entry{Graduate Researcher, PLUS Lab, Tel Aviv University}{Dec 2023 -- 2026}{}{}
\sub{Fingerprinting retrievers in black-box systems (M.Sc.\ thesis) [1]}
\begin{itemize}
  \item Built TellTail, a black-box \textbf{fingerprinting} attack that identifies the embedding model behind retrieval and RAG systems.
  \item Evaluated on \textbf{53 embedding retrievers} in an \textbf{open-world} setting (19 known, 34 unknown).
  \item Achieved \textbf{92.5\%} attack success rate (ASR) from LLM responses alone.
  \item Fingerprinted the retriever behind \textbf{OpenWebUI}, an open-source RAG deployment.
\end{itemize}
\sub{Forecasting cyber incidents with LLMs [2]}
\begin{itemize}
  \item Engineered an LLM pipeline turning news and podcasts into features to forecast \textbf{15,700+} security incidents.
  \item Ran \textbf{Internet-wide scans} on multiple ports with ZMap and built DNS infrastructure for network measurements.
  \item Reached \textbf{81.3\%} ROC AUC vs.\ 77.5\% for the prior state of the art; trained Longformer with multi-GPU DDP.
\end{itemize}
